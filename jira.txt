const schemas = [...document.querySelectorAll(
  'script[type="application/ld+json"]'
)];

schemas.map((script, index) => {
  try {
    const data = JSON.parse(script.textContent);
    return {
      index,
      validJSON: true,
      types: (data["@graph"] || [data])
        .map(item => item["@type"]),
      context: data["@context"]
    };
  } catch (error) {
    return {
      index,
      validJSON: false,
      error: error.message
    };
  }
});
