SELECT DISTINCT
    ResortId,
    PropertyID,
    PropertyName,
    PropertyTypeCode,
    Rating,
    CountryCode,
    SupplierCode,
    TouristTax
FROM Ski.dbo.OptimisedSearch
WHERE SupplierCode = 'KAL'
  AND ResortId IN (1605, 3306)
ORDER BY ResortId, PropertyID;

SELECT *
FROM Ski.dbo.TouristTax
WHERE ResortId IN (1605, 3306);
