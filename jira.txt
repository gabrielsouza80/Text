import { test, expect } from '../../support/apiBaseTest';
import testData from '../../fixtures/datajson/dataAPI.json';
import * as http from 'node:http';
import * as https from 'node:https';

// =====================================================
// SKIP-2402 - BaggageAllowance Error Handling
// =====================================================
//
// Root cause:
// Kestrel rejected slow request bodies because of
// MinRequestBodyDataRate.
//
// The previous /Home/Error exception handler pointed
// to a missing endpoint, masking the original error.
//
// Expected behavior:
// - Valid requests return 200.
// - Invalid requests return 400.
// - Unexpected server exceptions return JSON 500.
// - Slow valid requests are processed successfully.
//
// IMPORTANT:
// Tests 01-03 run against the existing API.
// Tests 04-05 require an authorized QA environment.
// =====================================================

const endpoint = '/Details/BaggageAllowance';

const validPayload = {
  holidayId: testData.holidays.validId,
  supplierCode: testData.suppliers.validCode,
  duration: testData.holidays.expected.duration,
};

// Prevent internal exception details from leaking.
function expectNoInternalError(body: string): void {
  expect(body).not.toContain('System.InvalidOperationException');
  expect(body).not.toContain('BadHttpRequestException');
  expect(body).not.toContain('System.Runtime.CompilerServices');
  expect(body).not.toContain('/Home/Error');
}

// Sends a valid HTTP body slowly, using Node.js directly.
// Playwright's normal request.post() does not provide
// control over the transmission speed of each chunk.
async function postSlowBody(
  targetUrl: string,
  payload: object,
  authorization?: string
): Promise<{
  status: number;
  contentType: string;
  body: string;
}> {
  const url = new URL(targetUrl);

  if (!['http:', 'https:'].includes(url.protocol)) {
    throw new Error('Only HTTP/HTTPS URLs are supported');
  }

  const transport = url.protocol === 'https:' ? https : http;
  const bytes = Buffer.from(JSON.stringify(payload));

  const headers: Record<string, string> = {
    'Content-Type': 'application/json',
    'Content-Length': String(bytes.length),
  };

  if (authorization) {
    headers.Authorization = authorization;
  }

  return new Promise((resolve, reject) => {
    let finished = false;
    let timer: ReturnType<typeof setTimeout> | undefined;
    let offset = 0;

    const request = transport.request(
      url,
      {
        method: 'POST',
        headers,
      },
      (response) => {
        const chunks: Buffer[] = [];

        response.on('data', (chunk: Buffer) => {
          chunks.push(Buffer.from(chunk));
        });

        response.on('error', reject);

        response.on('end', () => {
          finished = true;

          if (timer) {
            clearTimeout(timer);
          }

          resolve({
            status: response.statusCode ?? 0,
            contentType: String(
              response.headers['content-type'] ?? ''
            ),
            body: Buffer.concat(chunks).toString('utf8'),
          });
        });
      }
    );

    request.setTimeout(40000, () => {
      request.destroy(new Error('Slow request timed out'));
    });

    request.on('error', (error) => {
      finished = true;

      if (timer) {
        clearTimeout(timer);
      }

      reject(error);
    });

    function sendNextChunk(): void {
      if (finished || request.destroyed) {
        return;
      }

      if (offset >= bytes.length) {
        request.end();
        return;
      }

      const next = bytes.subarray(offset, offset + 8);

      request.write(next);
      offset += next.length;

      timer = setTimeout(sendNextChunk, 1200);
    }

    sendNextChunk();
  });
}

// =====================================================
// TEST CASES
// =====================================================

test.describe('SKIP-2402 - BaggageAllowance error handling', () => {

  test('SKIP2402-01 - Valid request returns 200',
    async ({ detailsApi }) => {

      await test.step('Send valid BaggageAllowance payload', async () => {

        const response = await detailsApi.post(
          endpoint,
          validPayload
        );

        const status = response.status();

        expect(
          status,
          `Expected 200 for valid payload, received ${status}`
        ).toBe(200);

        console.log(
          `[SKIP2402-01] Valid request: ${status}`
        );
      });
    }
  );

  test('SKIP2402-02 - Empty body returns 400',
    async ({ detailsApi }) => {

      await test.step('Send an empty request body', async () => {

        const response = await detailsApi.post(
          endpoint,
          {}
        );

        const status = response.status();
        const body = await response.text();

        expect(
          status,
          `Expected validation error 400, received ${status}. ` +
          `Response: ${body}`
        ).toBe(400);

        expectNoInternalError(body);

        console.log(
          `[SKIP2402-02] Empty body: ${status}`
        );
      });
    }
  );

  test('SKIP2402-03 - Malformed JSON returns 400',
    async ({ request }) => {

      await test.step('Send malformed JSON', async () => {

        const response = await request.post(endpoint, {
          headers: {
            'Content-Type': 'application/json',
          },
          data: '{"holidayId":',
          timeout: 15000,
          failOnStatusCode: false,
        });

        const status = response.status();
        const body = await response.text();

        expect(
          status,
          `Expected 400 for malformed JSON, received ${status}. ` +
          `Response: ${body}`
        ).toBe(400);

        expectNoInternalError(body);

        console.log(
          `[SKIP2402-03] Malformed JSON: ${status}`
        );
      });
    }
  );

  test('SKIP2402-04 - Unexpected server exception returns JSON 500',
    async ({ request }) => {

      const faultUrl = process.env.SKIP2402_FAULT_URL;

      test.skip(
        !faultUrl,
        'Requires a backend fault-injection endpoint in QA. ' +
        'A client-side HTTP mock cannot validate the real server handler.'
      );

      await test.step('Validate real server exception handling', async () => {

        const response = await request.post(faultUrl!, {
          data: validPayload,
          timeout: 20000,
          failOnStatusCode: false,
        });

        const status = response.status();
        const contentType =
          response.headers()['content-type'] ?? '';

        const body = await response.text();

        expect(
          status,
          `Expected HTTP 500, received ${status}. ` +
          `Response: ${body}`
        ).toBe(500);

        expect(contentType).toContain('application/json');

        expectNoInternalError(body);

        const json = JSON.parse(body);

        expect(json).toEqual({
          error: 'An unexpected error occurred.',
        });

        console.log(
          `[SKIP2402-04] Server exception: ${status}`
        );
      });
    }
  );

  test('SKIP2402-05 - Slow valid body is processed',
    async () => {

      const slowUrl = process.env.SKIP2402_SLOW_URL;

      test.skip(
        !slowUrl,
        'Requires a full QA endpoint URL for a controlled ' +
        'slow-body request against the real server.'
      );

      test.setTimeout(60000);

      await test.step('Send valid JSON in slow HTTP chunks', async () => {

        const response = await postSlowBody(
          slowUrl!,
          validPayload,
          process.env.SKIP2402_AUTHORIZATION
        );

        expect(
          response.status,
          `Expected 200 for a complete valid slow request, ` +
          `received ${response.status}. ` +
          `Response: ${response.body}`
        ).toBe(200);

        expectNoInternalError(response.body);

        console.log(
          `[SKIP2402-05] Slow body: ${response.status}`
        );
      });
    }
  );

});
