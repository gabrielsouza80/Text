SELECT DISTINCT TOP (50)
    os.PropertyId,
    os.SupplierPriceAvailableId,
    spa.SupplierFileId,
    spa.SupplierUnitId,
    spa.SupplierCode,
    spa.Price,
    spa.StartDate,
    spa.Duration,
    spa.BoardBasisCode,
    spa.LastUpdated
FROM [Ski].[dbo].[OptimisedSearch] AS os
INNER JOIN [Ski].[dbo].[SupplierPriceAvailable] AS spa
    ON os.SupplierPriceAvailableId = spa.SupplierPriceAvailableId
WHERE os.PropertyId = 40374
  AND spa.SupplierFileId = 414
  AND spa.StartDate >= '2026-12-19'
  AND spa.StartDate < '2026-12-20'
ORDER BY spa.LastUpdated DESC;
