DECLARE @HolidayId BIGINT = 202075020;

SELECT
    p.SupplierPriceAvailableId AS HolidayId,
    p.SupplierFileId,
    p.Price,
    CONVERT(VARCHAR(16), l.[Date], 120) AS PriceImported
FROM [Ski].[dbo].[SupplierPriceAvailable] p
OUTER APPLY (
    SELECT TOP 1 [Date]
    FROM [Ski].[dbo].[SupplierFileImportLog]
    WHERE SupplierFileId = p.SupplierFileId
      AND Successful = 1
    ORDER BY [Date] DESC
) l
WHERE p.SupplierPriceAvailableId = @HolidayId;
